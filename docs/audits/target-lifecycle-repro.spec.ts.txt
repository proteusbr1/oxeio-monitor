import { describe, it, expect, vi } from 'vitest';
import { TargetsService } from '../src/targets/targets.service';

const old = new Date('2026-09-24T10:00:00Z');
const now = new Date('2026-09-26T10:00:00Z');
function service(prisma: any) {
  return new TargetsService(prisma, { record: vi.fn().mockResolvedValue(undefined) } as any, {} as any);
}
describe('audit: target lifecycle invariants', () => {
  it('does not accept two completions when only one daily slot remains', async () => {
    let completed = 24;
    let reads = 0;
    let release!: () => void;
    const gate = new Promise<void>(resolve => { release = resolve; });
    const prisma = {
      employee: { findUnique: vi.fn().mockResolvedValue({ staffType: 'designer', dailyDesignTarget: 25, policy: null }) },
      designTarget: {
        count: vi.fn(async () => { const snapshot = completed; if (++reads === 2) release(); await gate; return snapshot; }),
        updateMany: vi.fn(async () => { completed++; return { count: 1 }; }),
      },
    };
    const svc = service(prisma);
    vi.spyOn(svc as any, 'topUp').mockResolvedValue(undefined);
    await Promise.allSettled([svc.markDone(1, 101, 1, now), svc.markDone(1, 102, 1, now)]);
    expect(completed).toBeLessThanOrEqual(25);
  });
  it('clears prior upload and review pipeline when returning a design to the pool', async () => {
    const row: any = { status: 'done', checkedAt: old, checkedById: 2, errorFoundAt: old, fixedAt: old, fixedById: 2, uploadedAt: old, liveAt: old, liveAsin: 'B012345678', completedAt: old };
    const svc = service({ designTarget: { updateMany: vi.fn(async ({ data }) => { Object.assign(row, data); return { count: 1 }; }) } });
    await svc.update(101, 'pool', now, 1);
    expect(row.uploadedAt).toBeNull();
    expect(row.checkedAt).toBeNull();
    expect(row.liveAt).toBeNull();
  });
  it('preserves the original completion date on a repeated done request', async () => {
    const row: any = { status: 'done', completedAt: old, completedById: 2 };
    const svc = service({ designTarget: { updateMany: vi.fn(async ({ data }) => { Object.assign(row, data); return { count: 1 }; }) } });
    await svc.update(101, 'done', now, 1);
    expect(row.completedAt).toEqual(old);
  });
  it('does not delete work completed between selection and deletion', async () => {
    const row: any = { id: 101, status: 'assigned' };
    const svc = service({ designTarget: {
      findMany: vi.fn(async () => { const snapshot = { ...row }; row.status = 'done'; return [snapshot]; }),
      updateMany: vi.fn(async ({ where, data }) => {
        // A concurrent worker completed the row after the earlier read.
        if (where.status?.notIn?.includes(row.status)) return { count: 0 };
        Object.assign(row, data); return { count: 1 };
      }),
    } });
    await svc.softDelete([101], 1, '127.0.0.1', 'not_found');
    expect(row.status).toBe('done');
  });
});
